        PagamentoController.prototype.pesquisar = function() {

	        this.$scope.ngTable = this.grid();

        PagamentoController.prototype.entidade = function(id) {
            this.appRouteService.goTo('.entidade')
        }

        PagamentoController.prototype.detalharOrdemBancaria = function(ordemBancaria) {
        	var $this = this;
        	$this.filtroData.pagamento = $this.getPagamentoByOrdemBancaria(ordemBancaria);
        	this.detalhadaService.filtroParams = $this.filtroData;
            this.appRouteService.goTo('.detalharOrdemBancaria')
        }

        PagamentoController.prototype.detalharDesconto = function(processo) {
        	var $this = this;
            $this.filtroData.pagamento = $this.getPagamentoByNumeroProcesso(processo);
        	this.detalhadaService.filtroParams = $this.filtroData;
        	this.appRouteService.goTo('.detalharDesconto')
        }

        PagamentoController.prototype.detalharSaldo = function(banco, agencia, conta) {

        	var $this = this;
        	$this.filtroData.pagamento = $this.getPagamentoByContaCorrente(banco, agencia, conta);

        	if ($this.filtroData.geral) {
        		$this.filtroData.rotaOrigem = 'detalhada-multisselecao.acao.pagamento';
        	} else {
        		$this.filtroData.rotaOrigem = 'detalhada.acao.pagamento';
        	}

        	this.detalhadaService.filtroParams = $this.filtroData;
            this.appRouteService.goTo('.detalharSaldo')
        }
        
        PagamentoController.prototype.detalharExtratoConta = function(banco, agencia, conta) {

        	var $this = this;
        	$this.filtroData.pagamento = $this.getPagamentoByContaCorrente(banco, agencia, conta);

        	if ($this.filtroData.geral) {
        		$this.filtroData.rotaOrigem = 'detalhada-multisselecao.acao.pagamento';

        	}

        	this.detalhadaService.filtroParams = $this.filtroData;
            this.appRouteService.goTo('.detalharExtratoConta')
        }

        PagamentoController.prototype.voltar = function(id) {

        	if (this.filtroData.geral) {
        		this.appRouteService.goTo('detalhada-multisselecao.acao');
        	} else {
        		this.appRouteService.goTo('detalhada.acao');
        	}
        }

        PagamentoController.prototype.getPagamentoByOrdemBancaria = function(ordemBancaria) {

			var result = null;
			this.dados.forEach(function(obj) {
				if (obj.numeroDocumentoSiafi == ordemBancaria) {
					result = obj;
				}
			});
			return result;
		}
        
        PagamentoController.prototype.getPagamentoByNumeroProcesso = function(processo) {

			var result = null;
			this.dados.forEach(function(obj) {
				if (obj.id.processo == processo) {
					result = obj;
				}
			});
			return result;
		}

        PagamentoController.prototype.getPagamentoByContaCorrente = function(banco, agencia, conta) {

			var result = null;
			this.dados.forEach(function(obj) {
				if (obj.codigoBanco == banco && obj.codigoAgencia == agencia && obj.contaCorrente == conta) {
					result = obj;
				}
			});
			return result;
		}
