        UtilService.prototype.showBanner = function(funcionalidade) {
            var $this = this;

            $this.appModalService
            .setOptions({
                title: "BANNER",
                size: "md",
                scope: $this.$scope,
                backdrop: 'static',
                keyboard: false,
                templateUrl: 'app/pages/banner/views/modal-banner.tpl.html',
                buttons: {
                    "btn1": {
                        ngClick: function(){
                            $this.appModalService.close();
                        },
                        name: 'Fechar',
                        style: 'btn btn-primary'
                    }
                }
            }).open().then(
                function (sucesso) {
                	var naoExibirBanner = 'naoExibirBanner-' + funcionalidade;
                    if ($('#nao-exibir-novamente').is(':checked')) {
                        localStorage.setItem(naoExibirBanner, true);
                    } else {
                        localStorage.setItem(naoExibirBanner, false);
                    }
                },
                function (error) {

                    } else {
                        localStorage.setItem(naoExibirBanner, false);

                });

            setTimeout(function(){ $('.image-banner').closest('.modal-content').find('.modal-header').remove(); }, 500);
        }

        UtilService.prototype.redirectEmptyParameters = function(params, rota) {
            if (typeof(params) == 'undefined' || params == null || Object.keys(params).length == 0) {
                this.appRouteService.goTo(rota, {});
                return true;
            }
            return false;
        }
