			var AcaoController = function($scope, $log, ngTableParams, $filter,
					detalhadaService, appRouteService, appValidatorService,
					consolidadaService, dirigenteService, downloadService,
					appModalService, utilService) {
				this.$scope = $scope;
				this.$log = $log;
				this.ngTableParams = ngTableParams;
				this.$filter = $filter;
				this.appRouteService = appRouteService;
				this.detalhadaService = detalhadaService;
				this.appValidatorService = appValidatorService;
				this.consolidadaService = consolidadaService;
				this.dirigenteService = dirigenteService;
				this.downloadService = downloadService;
				this.appModalService = appModalService;

				if (utilService.redirectEmptyParameters(this.detalhadaService.filtroParams, 'detalhada')) {return;}

				this.$scope.showGrid = false;
				this.boxGrid = false;
				this.filtroData = this.detalhadaService.filtroParams;
				this.dados = [];
				this.totalGeral = {};
				this.dirigente = {};
				this.totalBloco = {};
				this.totalGrupo = {};

				if (this.filtroData.populacao == undefined
						|| (this.filtroData.entidade.codigoMunicipioIBGE != this.filtroData.ibgePesquisado)) {
					var $this = this;
					if (this.filtroData.entidade != undefined) {
						this.consolidadaService
								.cGet(
										'populacaoCenso/'
												+ $this.filtroData.entidade.codigoMunicipioIBGE)
								.then(
										function(sucesso) {
											$this.filtroData.populacao = sucesso.resultado.populacao;
											$this.filtroData.anoCenso = sucesso.resultado.anoCenso;
											$this.filtroData.ibgePesquisado = $this.filtroData.entidade.codigoMunicipioIBGE;
										},
										function(erro) {
											$this.$scope.$appNotify.error(erro.data.mensagens);
										});
					}
				}

				if (this.filtroData.entidade != undefined
						&& (this.filtroData.tipoConsulta != undefined && this.filtroData.tipoConsulta.valor == 2)) {
					var $this = this;
					parametros = {
						esferaAdministrativa : $this.filtroData.entidade.esferaAdministrativa,
						codigoIbge : $this.filtroData.entidade.codigoMunicipioIBGE
					};

					var dirigenteLocal = this.dirigenteService
							.recuperarDirigenteLocal(
									parametros.esferaAdministrativa,
									parametros.codigoIbge);

					if (dirigenteLocal) {
						$this.dirigente = dirigenteLocal;
					} else {
						this.dirigenteService
								.recuperarDirigente(
										parametros.esferaAdministrativa,
										parametros.codigoIbge)
								.then(
										function(sucesso) {
											$this.dirigente = sucesso.resultado;
											$this.dirigente.codigoIbge = parametros.codigoIbge;
											$this.dirigente.esferaAdministrativa = parametros.esferaAdministrativa;
											$this.dirigente.noPresidenteConselho = $this.$filter('uppercase')($this.dirigente.noPresidenteConselho);
											$this.dirigenteService.dirigentes
													.push($this.dirigente);
										},
										function(erro) {
											$this.$scope.$appNotify.error(erro.data.mensagens);
										});
					}

				}

				this.pesquisar();

			}

														if ($this.filtroData.ano.valor < 2018) {

														} else {

															val['grupo'] = val.grupoAcao.nome;
														}
														if (val.valorTotal <= 0) {
															val['componente'] = "";
															val['acaoServicoEstrategia'] = "SEM REPASSE EM "

															val['valorTotal'] = "";
															val['valorDesconto'] = "";
															val['valorLiquido'] = "";
														} else {
															if (val.componenteBloco.nome) {
																val['componente'] = "{{'"
																	+ val.componenteBloco.nome
																	+ "'}}";
															} else {
																val['componente'] = "";
															}
															val['acaoServicoEstrategia'] = "{{'"

															val['valorTotal'] = "{{'"

														}

											},

												title : "{{ '"

														+ "' | number:2}}",
												visible : true

			AcaoController.prototype.entidade = function(id) {
				this.appRouteService.goTo('.entidade')
			}

			AcaoController.prototype.detalhar = function(coProgramaFundo, detalhaPagamento) {

				this.filtroData.programaFundo = this.getPagamentoByProgramaFundo(coProgramaFundo);

				this.detalhadaService.filtroParams = this.filtroData;

				if (detalhaPagamento) {
					this.appRouteService.goTo('.pagamento')
				} else {
					this.appRouteService.goTo('.conta')
				}
			}

			AcaoController.prototype.getPagamentoByProgramaFundo = function(
					coProgramaFundo) {

				var result = null;
				this.dados.forEach(function(obj) {
					if (obj.id == coProgramaFundo) {
						result = obj;
					}
				});
				return result;
			}

			AcaoController.prototype.voltar = function(id) {

				if (this.filtroData.geral) {
					this.appRouteService.goTo('detalhada-multisselecao');
				} else {
					this.appRouteService.goTo('detalhada');
				}
			}
			AcaoController.prototype.imprimir = function() {
				window.print();
			}

			AcaoController.prototype.gerarPlanilha = function() {
				var $this = this;

				$this.detalhadaService
						.gerarPlanilhaDetalhada(
								{
									ano : $this.filtroData.ano.valor,
									coMesAno : ($this.filtroData.mes != null ? $this.filtroData.mes.valor
											: ''),
									tipoConsulta : $this.filtroData.tipoConsulta != undefined ? $this.filtroData.tipoConsulta.valor
											: "",
									coBloco : ($this.filtroData.blocos != null ? $this.filtroData.blocos.id
											: ''),
									coComponente : ($this.filtroData.componentes != null ? $this.filtroData.componentes.id
											: ''),
									coGrupoAcao : ($this.filtroData.grupo != null ? $this.filtroData.grupo.id
											: ''),
									coPlanoOrcamentario : ($this.filtroData.planos != null ? $this.filtroData.planos.id
											: ''),
									coAcao : ($this.filtroData.acoes != null ? $this.filtroData.acoes.id
											: ''),
									sgUf : $this.filtroData.entidade.uf,
									coMunicipioIbge : $this.filtroData.entidade.codigoMunicipioIBGE,
									formaRepasse : ($this.filtroData.repasse != null ? $this.filtroData.repasse.valor
											: ''),
									dtInicioOb : ($this.filtroData.dataInicialOb != null ? moment(
											$this.filtroData.dataInicialOb)
											.format('YYYY-MM-DD')
											: ''),
									dtFinalOb : ($this.filtroData.dataFinalOb != null ? moment(
											$this.filtroData.dataFinalOb)
											.format('YYYY-MM-DD')
											: ''),
									nuAcaoJudicial : $this.filtroData.nuAcaoJudicial,
									nuCpfCnpjUg : $this.filtroData.entidade.cpfCnpj,
									nuProcesso : $this.filtroData.processo,
									nuProposta : $this.filtroData.proposta,
									nuPortaria : $this.filtroData.portaria,
									noRazaoSocial : $this.filtroData.entidade.razaoSocial,
									nuCnpj : $this.filtroData.entidade.cpfCnpjFormatado,
									noMunicipio : $this.filtroData.entidade.municipio
								})
						.then(
								function(response) {
									var blob = new Blob(
											[ response ],
											{
												type : "application/vnd.ms-excel; charset=UTF-8"
											});
									if(blob.size > 0){
										$this.downloadService.downloadXls(blob,
										'PlanilhaDetalhada');
									}else{
										$this.$scope.$appNotify.info("Nenhum detalhamento a ser apresentado.");
									}

								}, function(error) {
									$this.$scope.$appNotify.error(error.data.mensagens);
								});
			}
