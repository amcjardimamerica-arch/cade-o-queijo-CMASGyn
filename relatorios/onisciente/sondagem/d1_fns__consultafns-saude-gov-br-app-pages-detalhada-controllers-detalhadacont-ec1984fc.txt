        DetalhadaController.prototype.mudaEstado = function(siglaUf) {
            var $this = this;
            this.pesquisa.municipio = null;
            this.municipioService.buscarMunicipiosPorSiglaUF(siglaUf).then(function(sucesso) {

        DetalhadaController.prototype.mudaTipoConsulta = function(ano) {
            var $this = this;
            this.pesquisa.tipoConsulta = null;
            this.tipoConsultaService.listar($this.pesquisa.ano.descricao).then(function(sucesso) {
        		$this.tipoConsulta = sucesso.resultado;
        		$this.tipoConsulta.forEach(function(obj){
        			if (obj.valor == 2){
        				$this.pesquisa.tipoConsulta = obj;
        			}
        		});
        		$this.ajustarBlocoComponenteAcao();
            		}, function(erro) {
            				$this.$scope.$appNotify.error(erro.data.mensagens);
            	});
        }
        
        DetalhadaController.prototype.mudaBloco = function(coTipoConsulta, ano) {
            var $this = this;
            this.limparBlocos();
            var parametros = {
            		coTipoConsulta: coTipoConsulta,
            		ano: ano
            }
            
            var blocoLocal = this.blocoService.recuperarBlocoLocal(coTipoConsulta,ano);
            if (blocoLocal.length > 0 ) {
            	$this.blocos = blocoLocal;
            } else {
            	$this.blocoService.listarPorTipoConsultaEAno(parametros).then(function(sucesso) {
            		$this.blocos = sucesso.resultado;
            		$this.blocos.forEach(function(obj){
            			obj.coTipoConsulta = parametros.coTipoConsulta;
            			obj.ano = parametros.ano
            			$this.blocoService.blocosLocais.push(obj);
            		});
            	}, function(erro) {
            		$this.$scope.$appNotify.error(erro.data.mensagens);
            	});	
            }
            
        }
        
        DetalhadaController.prototype.carregaGrupoOuComponente = function(ano) {
        	var $this = this;
        	this.limparComponenteAcao();
        	if (ano > 2017){
        		this.carregaGrupoAcao(this.pesquisa.blocos, ano)
        	}else{
        		this.carregaComponentes(this.pesquisa.tipoConsulta.valor, this.pesquisa.blocos, ano)
        	}
        }
        
        DetalhadaController.prototype.carregaGrupoAcao = function(bloco, ano) {
            var $this = this;
            this.limparGrupoAcao();
            var parametros = {
            		bloco: this.pesquisa.blocos!=null ? this.pesquisa.blocos.id : null,
            		ano: ano
            }
            var grupoAcaoLocal = this.blocoService.recuperarGrupoAcaoLocal(parametros.bloco,parametros.ano);
            if (grupoAcaoLocal.length > 0 ) {
            	$this.grupo = grupoAcaoLocal;
            } else{
            	this.blocoService.listarGrupoAcaoPorBlocoEAno(parametros).then(function(sucesso) {
            		$this.grupo = sucesso.resultado;
            		$this.grupo.forEach(function(obj){
            			obj.bloco = parametros.bloco;
            			obj.ano = parametros.ano
            			$this.blocoService.gruposAcoesLocais.push(obj);
            		});
            	}, function(erro) {
            		$this.$scope.$appNotify.error(erro.data.mensagens);
            	});
            } 
        }

        DetalhadaController.prototype.carregaComponentes = function(coTipoConsulta, bloco, ano) {

        DetalhadaController.prototype.carregaAcoes = function() {

            			planoOrcamentario: this.pesquisa.planos!= null ? this.pesquisa.planos.id : null,

        DetalhadaController.prototype.grid = function() {
            var $this = this;
            return new this.ngTableParams(
                {
                    page: 1,
                    count: 10,

                    ano:$this.pesquisa.ano.valor,
                    mes:($this.pesquisa.mes!=null ? $this.pesquisa.mes.valor : ''),
					tipoConsulta:$this.pesquisa.tipoConsulta != undefined ? $this.pesquisa.tipoConsulta.valor : "",
					blocos:($this.pesquisa.blocos!=null ? $this.pesquisa.blocos.id : ''),

                },
                {
                    getData: function($defer, params) {

                    	var filtro = Object.assign({
                    		nuAcaoJudicial:$this.pesquisa.nuAcaoJudicial,
                    		cpfCnpjUg:$this.pesquisa.cpfCnpjUg,
                    		processo:$this.pesquisa.processo,
        					proposta:$this.pesquisa.proposta,
        					portaria:$this.pesquisa.portaria
						}, params.url());

                        var sucesso = function(response) {
                            params.total(response.resultado.total);

                            $this.dados = response.resultado.dados;

                            angular.forEach($this.dados, function (val, index) {
                            	var isFernandoNoronha = false;
                            	if (val.codigoMunicipioIBGE == '260545'){

                            	}

                                if (isFernandoNoronha){
                                	val['municipio'] = null;
                                }
                            });

                            var orderedData = params.sorting() && params.orderBy().length ?
                                    $this.$filter('orderBy')($this.dados, params.orderBy()) :
                                    $this.dados;

                        }
                        $this.detalhadaService.recuperarEntidades(filtro).then(sucesso,erro);
                    }
                });
        }

        DetalhadaController.prototype.pesquisar = function() {
            var $this = this;
            $this.appValidatorService.validate($this.$scope).success(function(sucesso) {

            	$this.$scope.showGrid = true;

                $this.boxGrid = true;
                $this.$scope.ngTable = $this.grid();

                $this.pesquisado = JSON.parse(JSON.stringify($this.pesquisa));
                
                if ($this.pesquisa.municipio != undefined ) {
                	  $this.consolidadaService.cGet('populacaoCenso/'+$this.pesquisa.municipio.coMunicipioIbge).then(function(sucesso) {
                          $this.pesquisado.populacao = sucesso.resultado.populacao;
                          $this.pesquisado.anoCenso = sucesso.resultado.anoCenso;
                      }, function(erro) {
                          $this.$scope.$appNotify.error(erro.data.mensagens);
                      });
                }

            }).error(function(erro){
               $this.$scope.$appNotify.error( $this.$filter('translate')('mensagem.obrigatorio.erro') );
            })
        }

        DetalhadaController.prototype.getEntidadeByCNPJ = function(cpfCnpj) {

			var result = null;
			this.dados.forEach(function(obj) {
				if (obj.cpfCnpj == cpfCnpj) {
					result = obj;
				}
			});
			return result;
		}

        DetalhadaController.prototype.acao = function(cpfCnpj) {

        	this.pesquisado.entidade = this.getEntidadeByCNPJ(cpfCnpj);
        	this.detalhadaService.filtroParams = this.pesquisado;
        	this.detalhadaService.filtroParams.geral = false;
        	this.appRouteService.goTo('.acao');
        }

        DetalhadaController.prototype.validarTipoConsulta = function(){

        	if ($("#cpfCnpjUg").val() || $("#processo").val() || $("#proposta").val()) {
        	    return false;
        	}else{
        	    return true;
        	}
        }

        DetalhadaController.prototype.ajustarBlocoComponenteAcao = function(){
        	this.pesquisa.acoes = {};
        	this.pesquisa.blocos = {};
        	this.pesquisa.componentes = {};
        	this.pesquisa.planos = {};
        	this.pesquisa.grupo = {}
        	this.grupo = {};
        	delete this.blocoService.gruposAcoesLocai;
        	this.acoes = {};
        	this.componentes = {};
        	this.planos = {};
        	if (this.pesquisa.ano != null && this.pesquisa.tipoConsulta != undefined && (this.pesquisa.tipoConsulta.valor == 1 || this.pesquisa.tipoConsulta.valor == 2 || this.pesquisa.tipoConsulta.valor == 3)){
        		this.mudaBloco(this.pesquisa.tipoConsulta.valor, this.pesquisa.ano.descricao);
        		if (this.pesquisa.ano.valor > 2017 && this.pesquisa.tipoConsulta.valor == 3){
        			this.carregaGrupoAcao(null,this.pesquisa.ano.descricao);
        		}
        	}
        }

        DetalhadaController.prototype.limparAcoes = function() {

        DetalhadaController.prototype.limparGrupoAcao = function() {

        DetalhadaController.prototype.limparPlanoOrcamentario = function() {
